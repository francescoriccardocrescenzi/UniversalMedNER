\begin{frame}[fragile]{Pipeline}
  Each run is identified by a \texttt{--label}, driving a five-stage pipeline:
  \begin{enumerate}
    \item Baseline evaluation (no adapter) $\rightarrow$ \texttt{baseline\_metrics.json}
    \item LoRA SFT training $\rightarrow$ pushed to the Hub as \texttt{<label>\_ner\_sft}
    \item Evaluate the SFT checkpoint $\rightarrow$ \texttt{sft\_metrics.json}
    \item LoRA GRPO training on top of SFT $\rightarrow$ pushed as \texttt{<label>\_ner\_grpo}
    \item Evaluate the GRPO checkpoint $\rightarrow$ \texttt{grpo\_metrics.json}
  \end{enumerate}
  \begin{exampleblock}{Run it}
    \texttt{src/run\_full\_ner\_pipeline.sh --label=my\_run}
  \end{exampleblock}
\end{frame}
